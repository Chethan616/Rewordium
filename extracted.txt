/**
 * Renders styled text (Meme or Bubbly) to a Bitmap that can be added
 * as an image overlay via PhotoEditor.addImage(). This avoids the
 * limitations of PhotoEditor's built-in text which doesn't support
 * outlined or pill-background text.
 */
private fun renderStyledTextBitmap(
    text: String,
    textColor: Int,
    fontSize: Float,
    style: StickerTextStyle,
): Bitmap {
    val density = 2f // Render at 2x for crispness
    val scaledFontSize = fontSize * density

    val fillPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = textColor
        textSize = scaledFontSize
        typeface = Typeface.DEFAULT_BOLD
        textAlign = Paint.Align.CENTER
    }

    // Measure text bounds
    val textBounds = android.graphics.Rect()
    fillPaint.getTextBounds(text, 0, text.length, textBounds)
    val textWidth = fillPaint.measureText(text)

    return when (style) {
        StickerTextStyle.Meme -> {
            val strokeWidth = scaledFontSize * 0.12f
            val padH = (strokeWidth * 2).toInt()
            val padV = (strokeWidth * 2).toInt()
            val bmpW = (textWidth + padH * 2).toInt().coerceAtLeast(1)
            val bmpH = (textBounds.height() + padV * 2).toInt().coerceAtLeast(1)
            val bmp = Bitmap.createBitmap(bmpW, bmpH, Bitmap.Config.ARGB_8888)
            val canvas = Canvas(bmp)
            canvas.drawColor(AndroidColor.TRANSPARENT, PorterDuff.Mode.CLEAR)

            val cx = bmpW / 2f
            val cy = bmpH / 2f + textBounds.height() / 2f - textBounds.bottom

            // Draw stroke (black outline)
            val strokePaint = Paint(fillPaint).apply {
                color = AndroidColor.BLACK
                this.style = Paint.Style.STROKE
                this.strokeWidth = strokeWidth
                strokeJoin = Paint.Join.ROUND
                strokeCap = Paint.Cap.ROUND
            }
            canvas.drawText(text, cx, cy, strokePaint)

            // Draw fill on top
            fillPaint.style = Paint.Style.FILL
            canvas.drawText(text, cx, cy, fillPaint)

            bmp
        }
        StickerTextStyle.Bubbly -> {
            val padH = (scaledFontSize * 0.5f).toInt()
            val padV = (scaledFontSize * 0.35f).toInt()
            val bmpW = (textWidth + padH * 2).toInt().coerceAtLeast(1)

/**
 * Produces a die-cut sticker outline by dilating the alpha channel of [src]
 * by [outlineWidth] pixels and drawing the dilated region in [outlineColor]
 * behind the original image. The result is a new bitmap where the subject
 * has a thick colored border like a real printed sticker.
 */
private fun addStickerOutline(
    src: Bitmap,
    outlineColor: Int = AndroidColor.WHITE,
    outlineWidth: Int = 12,
): Bitmap {
    val w = src.width
    val h = src.height
    // Expand canvas to accommodate the outline
    val expandedW = w + outlineWidth * 2
    val expandedH = h + outlineWidth * 2
    val result = Bitmap.createBitmap(expandedW, expandedH, Bitmap.Config.ARGB_8888)
    val canvas = Canvas(result)
    canvas.drawColor(AndroidColor.TRANSPARENT, PorterDuff.Mode.CLEAR)

    // Build an alpha mask from the source
    val alphaPixels = IntArray(w * h)
    src.getPixels(alphaPixels, 0, w, 0, 0, w, h)

    // Create dilated alpha mask
    val dilatedAlpha = BooleanArray(expandedW * expandedH)
    val radiusSq = outlineWidth * outlineWidth

    for (y in 0 until h) {
        for (x in 0 until w) {
            val alpha = (alphaPixels[y * w + x] ushr 24) and 0xFF
            if (alpha > 20) { // Threshold to skip near-transparent pixels
                // Mark all pixels within outlineWidth radius as dilated
                val cx = x + outlineWidth
                val cy = y + outlineWidth
                for (dy in -outlineWidth..outlineWidth) {
                    for (dx in -outlineWidth..outlineWidth) {
                        if (dx * dx + dy * dy <= radiusSq) {
                            val nx = cx + dx
                            val ny = cy + dy
                            if (nx in 0 until expandedW && ny in 0 until expandedH) {
                                dilatedAlpha[ny * expandedW + nx] = true
                            }
                        }
                    }
                }
            }
        }
    }

    // Draw outline from dilated mask
    val outlinePaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = outlineColor
        style = Paint.Style.FILL
    }
    val outlineBmp = Bitmap.createBitmap(expandedW, expandedH, Bitmap.Config.ARGB_8888)
    val outlinePixels = IntArray(expandedW * expandedH)
    for (i in dilatedAlpha.indices) {
        outlinePixels[i] = if (dilatedAlpha[i]) outlineColor else AndroidColor.TRANSPARENT
    }
    outlineBmp.setPixels(outlinePixels, 0, expandedW, 0, 0, expandedW, expandedH)
    canvas.drawBitmap(outlineBmp, 0f, 0f, null)
    outlineBmp.recycle()

    // Draw original image centered on top
    canvas.drawBitmap(src, outlineWidth.toFloat(), outlineWidth.toFloat(), null)

    return result
}
